analytics_query.sql 
WITH vendas_ranking AS (
  SELECT
    v.vendedor,
    v.produto,
    SUM(v.valor)  AS total,
    RANK() OVER (
      PARTITION BY v.regiao
      ORDER BY SUM(v.valor) DESC
    ) AS posicao
  FROM  fato_vendas v
  WHERE v.ano = 2026
  GROUP BY v.vendedor, v.produto, v.regiao
)
SELECT * FROM vendas_ranking
WHERE  posicao <= 5
-- Top 5 vendedores por região ✓
